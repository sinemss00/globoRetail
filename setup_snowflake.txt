CREATE ROLE IF NOT EXISTS GLOBO_ANALYTICS_ROLE;

CREATE WAREHOUSE IF NOT EXISTS GLOBO_WH
    WITH WAREHOUSE_SIZE = 'XSMALL'
    AUTO_SUSPEND = 180
    AUTO_RESUME = TRUE;

CREATE DATABASE IF NOT EXISTS GLOBO_RETAIL_DB;
USE DATABASE GLOBO_RETAIL_DB;

CREATE SCHEMA IF NOT EXISTS RAW;
CREATE SCHEMA IF NOT EXISTS CLEANSED;
CREATE SCHEMA IF NOT EXISTS STAR;
CREATE SCHEMA IF NOT EXISTS PRESENTATION;

CREATE OR REPLACE FILE FORMAT STAGING_PARQUET_FORMAT
    TYPE = PARQUET;

CREATE OR REPLACE FILE FORMAT STAGING_CSV_FORMAT
    TYPE = CSV
    FIELD_DELIMITER = ','
    SKIP_HEADER = 1;

CREATE OR REPLACE STAGE RAW.S3_PROCESSED_STAGE
    URL = 's3://globo-retail-analytics-2026/processed/'
    FILE_FORMAT = STAGING_PARQUET_FORMAT;

CREATE OR REPLACE TABLE CLEANSED.SALES_CLEAN (
    ORDER_ID STRING,
    CUSTOMER_ID STRING,
    PRODUCT_ID STRING,
    PRODUCT_NAME STRING,
    CATEGORY STRING,
    BRAND STRING,
    REGION STRING,
    ORDER_DATE DATE,
    QUANTITY INT,
    UNIT_PRICE DECIMAL(10, 2),
    TOTAL_AMOUNT DECIMAL(10, 2),
    MARGIN DECIMAL(10, 2)
);

CREATE OR REPLACE TABLE STAR.DIM_DATE AS
SELECT DISTINCT
    ORDER_DATE AS DATE_KEY,
    EXTRACT(YEAR FROM ORDER_DATE) AS YEAR,
    EXTRACT(MONTH FROM ORDER_DATE) AS MONTH,
    EXTRACT(DAY FROM ORDER_DATE) AS DAY,
    EXTRACT(QUARTER FROM ORDER_DATE) AS QUARTER
FROM CLEANSED.SALES_CLEAN;

CREATE OR REPLACE TABLE STAR.DIM_PRODUCT AS
SELECT DISTINCT
    PRODUCT_ID AS PRODUCT_KEY,
    PRODUCT_NAME,
    CATEGORY,
    BRAND
FROM CLEANSED.SALES_CLEAN;

CREATE OR REPLACE TABLE STAR.FACT_SALES AS
SELECT
    ORDER_ID,
    CUSTOMER_ID,
    PRODUCT_ID AS PRODUCT_KEY,
    ORDER_DATE AS DATE_KEY,
    REGION,
    QUANTITY,
    UNIT_PRICE,
    TOTAL_AMOUNT,
    MARGIN
FROM CLEANSED.SALES_CLEAN;

DROP MATERIALIZED VIEW IF EXISTS PRESENTATION.MV_AVG_ORDER_VALUE_BY_REGION;
DROP MATERIALIZED VIEW IF EXISTS PRESENTATION.MV_TOP_BRANDS_BY_QUANTITY;
DROP MATERIALIZED VIEW IF EXISTS PRESENTATION.MV_MONTHLY_REVENUE_GROWTH;
DROP MATERIALIZED VIEW IF EXISTS PRESENTATION.MV_CATEGORY_MARGIN_ANALYSIS;

CREATE OR REPLACE VIEW PRESENTATION.MV_AVG_ORDER_VALUE_BY_REGION AS
SELECT
    REGION,
    ROUND(AVG(TOTAL_AMOUNT), 2) AS AVG_ORDER_VALUE,
    COUNT(ORDER_ID) AS TOTAL_ORDERS
FROM CLEANSED.SALES_CLEAN
GROUP BY REGION;

CREATE OR REPLACE VIEW PRESENTATION.MV_TOP_BRANDS_BY_QUANTITY AS
SELECT
    BRAND,
    SUM(QUANTITY) AS TOTAL_QUANTITY_SOLD
FROM CLEANSED.SALES_CLEAN
GROUP BY BRAND;

CREATE OR REPLACE VIEW PRESENTATION.MV_MONTHLY_REVENUE_GROWTH AS
SELECT
    EXTRACT(YEAR FROM ORDER_DATE) AS YEAR,
    EXTRACT(MONTH FROM ORDER_DATE) AS MONTH,
    SUM(TOTAL_AMOUNT) AS TOTAL_REVENUE
FROM CLEANSED.SALES_CLEAN
GROUP BY EXTRACT(YEAR FROM ORDER_DATE), EXTRACT(MONTH FROM ORDER_DATE);

CREATE OR REPLACE VIEW PRESENTATION.MV_CATEGORY_MARGIN_ANALYSIS AS
SELECT
    CATEGORY,
    SUM(TOTAL_AMOUNT) AS TOTAL_REVENUE,
    SUM(MARGIN) AS TOTAL_MARGIN,
    ROUND(AVG(MARGIN), 2) AS AVG_MARGIN
FROM CLEANSED.SALES_CLEAN
GROUP BY CATEGORY;
